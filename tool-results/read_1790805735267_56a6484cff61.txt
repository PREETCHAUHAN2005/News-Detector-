    74→/** Aggregate daily rows into weekly buckets for the wider windows. */
    75→function bucketWeekly(
    76→  daily: { date: string; total: number; contradicted: number }[]
    77→): { date: string; total: number; contradicted: number }[] {
    78→  const weeks = new Map<string, { total: number; contradicted: number }>();
    79→  for (const day of daily) {
    80→    const d = new Date(`${day.date}T00:00:00`);
    81→    // Monday-based week start
    82→    const weekday = (d.getDay() + 6) % 7;
    83→    const monday = new Date(d);
    84→    monday.setDate(d.getDate() - weekday);
    85→    const key = monday.toISOString().slice(0, 10);
    86→    const entry = weeks.get(key) ?? { total: 0, contradicted: 0 };
    87→    entry.total += day.total;
    88→    entry.contradicted += day.contradicted;
    89→    weeks.set(key, entry);
    90→  }
    91→  return Array.from(weeks.entries())
    92→    .sort(([a], [b]) => a.localeCompare(b))
    93→    .map(([date, v]) => ({ date, ...v }));
    94→}

   136→/* ───────────────────────── chart tooltip ───────────────────────── */
   137→
   138→interface TooltipPayloadItem {
   139→  name?: string | number;
   140→  value?: number | string;
   141→  color?: string;
   142→  payload?: Record<string, unknown>;
   143→}
   144→
   145→function ChartTooltip({
   146→  active,
   147→  items,
   148→  fmt,
   149→}: {
   150→  active?: boolean;
   151→  items: TooltipPayloadItem[] | undefined;
   152→  fmt: (item: TooltipPayloadItem) => { label: string; value: string } | null;
   153→}) {
   154→  if (!active || !items?.length) return null;
   155→  const rows = items.map(fmt).filter((x): x is { label: string; value: string } => x !== null);
   156→  if (!rows.length) return null;
   157→  return (
   158→    <div className="rounded-lg border border-zinc-800 bg-zinc-950/95 px-3 py-2 shadow-xl">
   159→      {rows.map((r, i) => (
   160→        <p key={`${r.label}-${i}`} className="text-xs text-zinc-300">
   161→          <span className="mr-2 inline-block size-2 rounded-full align-middle" style={{ background: items[i]?.color ?? "#a1a1aa" }} aria-hidden />
   162→          <span className="text-zinc-400">{r.label}</span>{" "}
   163→          <span className="font-mono tabular-nums text-zinc-100">{r.value}</span>
   164→        </p>
   165→      ))}
   166→    </div>
   167→  );
   168→}
