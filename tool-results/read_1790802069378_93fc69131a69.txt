    98→export function AnalysisView() {
    99→  const activeAnalysisId = useAppStore((s) => s.activeAnalysisId);
   100→  const setView = useAppStore((s) => s.setView);
   101→  const { data: session } = useSession();
   102→  const user = session?.user as SessionUser | undefined;
   103→  const role = user?.role ?? "viewer";

   105→  const isSenior = SENIOR_ROLES.includes(role);
   106→

   108→    queryKey: ["analysis", activeAnalysisId],

   112→
   113→  if (!activeAnalysisId) {

   118→        <Button variant="outline" size="sm" onClick={() => setView("home")}>
   119→          <ArrowLeft /> Back to home
   120→        </Button>
   121→      </div>
   122→    );
   123→  }

   125→  if (isLoading) {
   126→    return <AnalysisSkeleton />;

   474→/* ───────────────────── Reviewer decision panel ───────────────────── */
   475→
   476→function ReviewerDecisionPanel({
   477→  claim,
   478→  verdict,
   479→  analysis,
   480→  isSenior,
   481→}: {
   482→  claim: ClaimDTO;
   483→  verdict: VerdictDTO;
   484→  analysis: AnalysisDTO;
   485→  isSenior: boolean;
   486→}) {
   487→  const reviews = useMemo(
   488→    () => [...verdict.reviews].sort((a, b) => a.round - b.round),
   489→    [verdict.reviews]
   490→  );
   491→  const eff = effectiveLabel(claim);
   492→
   493→  const firstRound = reviews.length === 0;
   494→  const awaitingSecond = analysis.highImpact && reviews.length === 1;
   495→  const seniorTurn = awaitingSecond && isSenior;
   496→
   497→  return (
   498→    <section
   499→      aria-label="Reviewer decision"
   500→      className="rounded-lg border border-zinc-700/70 bg-zinc-900/60 p-4"
   501→    >
   502→      <div className="mb-3 flex flex-wrap items-center gap-x-2 gap-y-1">
   503→        <h4 className="flex items-center gap-1.5 text-xs font-semibold uppercase tracking-wide text-zinc-300">
   504→          <Gavel className="size-3.5 text-emerald-400" aria-hidden />
   505→          Reviewer decision
   506→        </h4>
   507→        <span className="text-[11px] text-zinc-500">
   508→          BR-11 — the human decision is final and overrides the model
   509→        </span>
   510→      </div>
   511→
   512→      {/* human verdict banner */}
   513→      {reviews.length > 0 && (
