   200→/* ───────────────────────── Claim navigator ───────────────────────── */
   201→
   202→/**
   203→ * ClaimNavigator — sticky mini-TOC for multi-claim dossiers. One chip per
   204→ * claim with a verdict-coloured dot; the chip scrolls to the claim and the
   205→ * active chip tracks scroll position. Evidence-first ordering is untouched:
   206→ * this only aids navigation within the dossier (BR-12 ergonomics).
   207→ */
   208→function ClaimNavigator({ claims }: { claims: ClaimDTO[] }) {
   209→  const [active, setActive] = useState<number>(0);
   210→
   211→  useEffect(() => {
   212→    const scroller = document.getElementById("main-content");
   213→    if (!scroller) return;
   214→
   215→    let raf = 0;
   216→    const onScroll = () => {
   217→      if (raf) return;
   218→      raf = requestAnimationFrame(() => {
   219→        raf = 0;
   220→        const scrollerRect = scroller.getBoundingClientRect();
   221→        let best = 0;
   222→        let bestDistance = Infinity;
   223→        for (const claim of claims) {
   224→          const el = document.getElementById(`claim-${claim.position}`);
   225→          if (!el) continue;
   226→          const distance = Math.abs(el.getBoundingClientRect().top - scrollerRect.top - 12);
   227→          if (distance < bestDistance) {
   228→            bestDistance = distance;
   229→            best = claim.position;
   230→          }
   231→        }
   232→        setActive(best);
   233→      });
   234→    };
   235→
   236→    scroller.addEventListener("scroll", onScroll, { passive: true });
   237→    onScroll();
   238→    return () => {
   239→      scroller.removeEventListener("scroll", onScroll);
   240→      if (raf) cancelAnimationFrame(raf);
   241→    };
   242→  }, [claims]);
   243→
   244→  const scrollToClaim = (position: number | null) => {
   245→    if (position === null) {
   246→      document.getElementById("main-content")?.scrollTo({ top: 0, behavior: "smooth" });
   247→      return;
   248→    }
   249→    document
   250→      .getElementById(`claim-${position}`)
   251→      ?.scrollIntoView({ behavior: "smooth", block: "start" });
   252→  };
   253→
   254→  return (
   255→    <nav
   256→      aria-label="Claim navigator"
   257→      className="sticky top-0 z-10 -mx-4 mt-4 border-b border-zinc-800/70 bg-zinc-950/90 px-4 py-2 backdrop-blur sm:-mx-6 sm:px-6"
   258→    >
   259→      <div className="flex items-center gap-2 overflow-x-auto pb-0.5 [scrollbar-width:none] [&::-webkit-scrollbar]:hidden">
   260→        <span className="flex shrink-0 items-center gap-1.5 pr-1 text-[10px] font-semibold uppercase tracking-wider text-zinc-500">
   261→          <ListTree className="size-3.5" aria-hidden />
   262→          {claims.length} claims
   263→        </span>
   264→        <button
   265→          type="button"
   266→          onClick={() => scrollToClaim(null)}
   267→          title="Scroll to the dossier overview"
   268→          className="shrink-0 rounded-full border border-zinc-800 bg-zinc-900/60 px-2.5 py-1 text-[11px] font-medium text-zinc-500 transition-colors hover:border-zinc-600 hover:text-zinc-200 focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-emerald-500/40 active:scale-[0.97]"
   269→        >
   270→          Overview
   271→        </button>
   272→        {claims.map((claim) => {
   273→          const label = claim.verdict?.label ?? null;
   274→          const meta = label && isVerdictLabel(label) ? LABEL_META[label] : null;
   275→          const isActive = active === claim.position;
   276→          return (
   277→            <button
   278→              key={claim.id}
   279→              type="button"
   280→              onClick={() => scrollToClaim(claim.position)}
   281→              aria-current={isActive ? "true" : undefined}
   282→              className={cn(
   283→                "flex shrink-0 items-center gap-1.5 rounded-full border px-2.5 py-1 text-[11px] font-medium transition-colors focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-emerald-500/40 active:scale-[0.97]",
   284→                isActive
   285→                  ? "border-zinc-600 bg-zinc-800/80 text-zinc-100 shadow-sm"
   286→                  : "border-zinc-800 bg-zinc-900/60 text-zinc-500 hover:border-zinc-600 hover:text-zinc-200"
   287→              )}
   288→            >
   289→              <span className="font-mono text-[10px] text-zinc-500">
   290→                #{String(claim.position + 1).padStart(2, "0")}
   291→              </span>
   292→              <span
   293→                aria-hidden
   294→                className={cn(
   295→                  "size-1.5 rounded-full",
   296→                  meta ? meta.dot : "bg-zinc-600"
   297→                )}
   298→              />
   299→              <span className="capitalize">{claim.claimType}</span>
   300→              {!claim.assessable && (
   301→                <span className="text-[10px] text-zinc-600">(excl.)</span>
   302→              )}
   303→            </button>
   304→          );
   305→        })}
   306→      </div>
   307→    </nav>
   308→  );
   309→}
